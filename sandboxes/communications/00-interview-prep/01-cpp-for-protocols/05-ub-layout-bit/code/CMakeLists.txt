cmake_minimum_required(VERSION 3.16)
project(es_pack05_ub_layout_bit LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20 CACHE STRING "C++ standard (20 recommended: <bit>; 17 skips the std:: cross-checks)")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
endif()

option(USE_SOLUTIONS     "Build against solutions/ instead of the stubs in include/" OFF)
option(ENABLE_SANITIZERS "Compile with -fsanitize=address,undefined (MSVC: /fsanitize=address)" ON)
option(BUILD_LABS        "Build the UB demonstration programs (no sanitizers; -O0 and -O2 variants)" ON)

if(USE_SOLUTIONS)
  set(ES_HEADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/solutions)
else()
  set(ES_HEADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/include)
endif()
message(STATUS "Headers from: ${ES_HEADER_DIR}")

if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
FetchContent_Declare(googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

enable_testing()
include(GoogleTest)

function(es_configure target)
  target_include_directories(${target} PRIVATE ${ES_HEADER_DIR})
  target_link_libraries(${target} PRIVATE GTest::gtest_main)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
    if(ENABLE_SANITIZERS)
      target_compile_options(${target} PRIVATE /fsanitize=address)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -g -fno-omit-frame-pointer)
    if(ENABLE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  gtest_discover_tests(${target} PROPERTIES TIMEOUT 30)
endfunction()

add_executable(pack05_tests
  tests/test_ub_safe.cpp
  tests/test_layout.cpp
  tests/test_bits.cpp)
es_configure(pack05_tests)

# ---- UB labs: run by hand, compare the -O0 and -O2 outputs (worksheet 3.1-3.3, Q8-Q10) ----
if(BUILD_LABS)
  foreach(lab lab_ub_check lab_shift lab_alias)
    foreach(opt O0 O2)
      add_executable(${lab}_${opt} labs/${lab}.cpp)
      if(MSVC)
        if(opt STREQUAL "O0")
          target_compile_options(${lab}_${opt} PRIVATE /W4 /Od)
        else()
          target_compile_options(${lab}_${opt} PRIVATE /W4 /O2)
        endif()
      else()
        target_compile_options(${lab}_${opt} PRIVATE -Wall -Wextra -g -${opt})
      endif()
    endforeach()
  endforeach()
endif()
